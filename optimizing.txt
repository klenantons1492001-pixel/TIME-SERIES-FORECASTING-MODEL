"""
Advanced Time Series Forecasting with Deep Learning and Explainability

Features:
- Synthetic multivariate time series (5 features, 5000+ observations)
- Trend, seasonality, and noise
- LSTM deep learning model
- SARIMAX statistical baseline
- RMSE and MAE comparison
- Integrated Gradients explainability
- Production-quality, modular code with docstrings
"""

import numpy as np
import pandas as pd
import torch
import torch.nn as nn

from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error, mean_absolute_error
from statsmodels.tsa.statespace.sarimax import SARIMAX


# =========================================================
# 1. Synthetic Dataset Generation
# =========================================================
def generate_synthetic_dataset(n_steps: int = 5000, seed: int = 42) -> pd.DataFrame:
    """
    Generate a synthetic multivariate time-series dataset with
    trend, seasonality, and noise.
    """
    np.random.seed(seed)
    t = np.arange(n_steps)

    trend = 0.002 * t
    daily = np.sin(2 * np.pi * t / 24)
    weekly = np.sin(2 * np.pi * t / 168)

    cpu_usage = 50 + trend + 10 * daily + np.random.normal(0, 2, n_steps)
    memory_usage = 60 + trend + 8 * weekly + np.random.normal(0, 2, n_steps)
    disk_io = 40 + 0.5 * cpu_usage + np.random.normal(0, 3, n_steps)
    network_io = 30 + 0.3 * memory_usage + np.random.normal(0, 2, n_steps)
    latency = 100 + 0.8 * cpu_usage + 0.5 * network_io + np.random.normal(0, 5, n_steps)

    return pd.DataFrame({
        "cpu_usage": cpu_usage,
        "memory_usage": memory_usage,
        "disk_io": disk_io,
        "network_io": network_io,
        "latency": latency
    })


# =========================================================
# 2. Data Preparation
# =========================================================
def create_sequences(data: np.ndarray, window: int):
    """
    Convert time-series data into supervised learning sequences.
    """
    X, y = [], []
    for i in range(len(data) - window):
        X.append(data[i:i + window])
        y.append(data[i + window, 0])  # Predict cpu_usage
    return np.array(X), np.array(y)


# =========================================================
# 3. LSTM Model Definition
# =========================================================
class LSTMForecaster(nn.Module):
    """
    Multivariate LSTM forecasting model.
    """
    def __init__(self, input_size: int, hidden_size: int = 64):
        super().__init__()
        self.lstm = nn.LSTM(input_size, hidden_size, batch_first=True)
        self.fc = nn.Linear(hidden_size, 1)

    def forward(self, x):
        out, _ = self.lstm(x)
        return self.fc(out[:, -1, :])


# =========================================================
# 4. Integrated Gradients Explainability
# =========================================================
def integrated_gradients(model, input_tensor, baseline=None, steps=50):
    """
    Compute Integrated Gradients for sequence models.
    """
    if baseline is None:
        baseline = torch.zeros_like(input_tensor)

    grads = []
    for alpha in np.linspace(0, 1, steps):
        interpolated = baseline + alpha * (input_tensor - baseline)
        interpolated.requires_grad_(True)
        output = model(interpolated)
        output.backward(torch.ones_like(output))
        grads.append(interpolated.grad.detach().numpy())

    avg_grads = np.mean(grads, axis=0)
    return (input_tensor.detach().numpy() - baseline.numpy()) * avg_grads


# =========================================================
# 5. Main Execution
# =========================================================
def main():
    # Generate dataset
    df = generate_synthetic_dataset()
    print("Dataset shape:", df.shape)

    # Scale data
    scaler = MinMaxScaler()
    scaled_data = scaler.fit_transform(df.values)

    # Create sequences
    WINDOW = 48
    X, y = create_sequences(scaled_data, WINDOW)

    split = int(0.8 * len(X))
    X_train, X_test = X[:split], X[split:]
    y_train, y_test = y[:split], y[split:]

    # Convert to tensors
    X_train_t = torch.tensor(X_train, dtype=torch.float32)
    y_train_t = torch.tensor(y_train, dtype=torch.float32).unsqueeze(1)
    X_test_t = torch.tensor(X_test, dtype=torch.float32)

    # Train LSTM
    model = LSTMForecaster(input_size=X.shape[2])
    criterion = nn.MSELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)

    print("\nTraining LSTM...")
    for epoch in range(40):
        optimizer.zero_grad()
        preds = model(X_train_t)
        loss = criterion(preds, y_train_t)
        loss.backward()
        optimizer.step()

        if epoch % 10 == 0:
            print(f"Epoch {epoch} | Loss: {loss.item():.4f}")

    # Evaluate LSTM
    model.eval()
    with torch.no_grad():
        dl_preds = model(X_test_t).numpy()

    dl_rmse = np.sqrt(mean_squared_error(y_test, dl_preds))
    dl_mae = mean_absolute_error(y_test, dl_preds)

    # SARIMAX Baseline
    print("\nTraining SARIMAX...")
    sarimax = SARIMAX(
        df["cpu_usage"],
        order=(1, 1, 1),
        seasonal_order=(1, 1, 1, 24)
    )
    sarimax_fit = sarimax.fit(disp=False)

    sarimax_preds = sarimax_fit.forecast(steps=len(y_test))

    sarimax_rmse = np.sqrt(
        mean_squared_error(df["cpu_usage"][-len(y_test):], sarimax_preds)
    )
    sarimax_mae = mean_absolute_error(
        df["cpu_usage"][-len(y_test):], sarimax_preds
    )

    # Explainability
    sample_input = X_test_t[:1]
    attributions = integrated_gradients(model, sample_input)
    feature_importance = attributions.mean(axis=1).squeeze()

    # Results
    print("\n=== MODEL COMPARISON ===")
    print(f"LSTM     -> RMSE: {dl_rmse:.4f}, MAE: {dl_mae:.4f}")
    print(f"SARIMAX  -> RMSE: {sarimax_rmse:.4f}, MAE: {sarimax_mae:.4f}")

    print("\n=== Integrated Gradients Feature Importance ===")
    for name, value in zip(df.columns, feature_importance):
        print(f"{name}: {value:.4f}")


# =========================================================
# Entry Point
# =========================================================
if __name__ == "__main__":
    main()
